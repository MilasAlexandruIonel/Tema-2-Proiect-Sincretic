#include <LiquidCrystal.h>
#include <avr/io.h>
#include <avr/interrupt.h>

const int rs = 12, en = 11, d4 = 5, d5 = 4, d6 = 3, d7 = 2;
LiquidCrystal lcd(rs, en, d4, d5, d6, d7);

int ore = 12;
int minute = 30;
int secunde = 0;
bool actualizeazaEcran = false;

void setup() {
  lcd.begin(16, 2);
  
  lcd.setCursor(0, 0);
  lcd.print("Pornire sistem..");
  delay(1000);
  lcd.clear();

  cli();

  TCCR1A = 0;
  TCCR1B = 0;
  TCNT1  = 0;
  OCR1A = 62499;
  TCCR1B |= (1 << WGM12);
  TCCR1B |= (1 << CS12);  
  TIMSK1 |= (1 << OCIE1A);

  ADMUX = (1 << REFS1) | (1 << REFS0);
  ADCSRA = (1 << ADEN) | (1 << ADPS2) | (1 << ADPS1) | (1 << ADPS0);

  sei();
}

ISR(TIMER1_COMPA_vect) {
  secunde++;
  if (secunde >= 60) {
    secunde = 0;
    minute++;
    if (minute >= 60) {
      minute = 0;
      ore++;
      if (ore >= 24) {
        ore = 0;
      }
    }
  }
  actualizeazaEcran = true;
}

float citesteTemperaturaLM35() {
  ADCSRA |= (1 << ADSC);
  while (ADCSRA & (1 << ADSC));
  int valoareADC = ADC;
  float temp = (valoareADC * 1.1 * 100.0) / 1024.0;
  return temp;
}

void loop() {
  if (actualizeazaEcran) {
    actualizeazaEcran = false;
    
    float temperatura = citesteTemperaturaLM35();
    
    lcd.setCursor(0, 0);
    lcd.print("Temp: ");
    lcd.print(temperatura, 1);
    lcd.print(char(223));
    lcd.print("C    ");

    lcd.setCursor(0, 1);
    lcd.print("Ceas: ");
    
    if(ore < 10) lcd.print('0');
    lcd.print(ore);
    lcd.print(':');
    if(minute < 10) lcd.print('0');
    lcd.print(minute);
    lcd.print(':');
    if(secunde < 10) lcd.print('0');
    lcd.print(secunde);
    lcd.print("    "); 
  }
}
